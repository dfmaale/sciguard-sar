		},

		\coordinate (P2) at (\DX,0);
		\coordinate (P3) at (2*\DX,0);

		\coordinate (P6) at (2*\DX,-\DY);
		
		% ----------------------------------------------------------------
		% 1. Training data and model
		% ----------------------------------------------------------------
		\node[panel=sgBlueBorder,anchor=north west] (box1) at (P1) {};
		\node[header,fill=sgNavy,anchor=north west] at (P1)
		{Training Data and Model};
		\node[number=sgNavy] at ($(P1)+(2.625cm,.16cm)$) {1};

		{$\mathcal D_n=\{(X_i,Y_i)\}_{i=1}^{n}$\\[3pt]
			$g=\mathcal I\circ f$};
		\node[explanation] at ($(P1)+(2.625cm,-3.35cm)$)
		{Fit one prediction pipeline using data from the source population.};

		% 2. Deployment environments

		\node[explanation] at ($(P2)+(2.625cm,-3.65cm)$)
		{Each point in $\mathcal E$ represents one possible deployment condition.};

		% 3. Risk under each environment

		\node[header,fill=sgGreen,anchor=north west] at (P5)

		\node[number=sgGreenDark] at ($(P5)+(2.625cm,.16cm)$) {5};

		{Only statistically supported environments are certified.};

	\caption{SciGuard statistical assurance
		framework. A predictive pipeline is evaluated over deployment environments
		indexed by missingness severity $m$ and distribution-shift severity $s$.
		Environment-specific risks and simultaneous upper confidence bounds are used
		to construct a conservative certified assurance region. The resulting map
		is schematic rather than an empirical result and identifies deployment conditions under which the model is statistically
		supported to satisfy the prespecified risk threshold $\tau$.}
